<script>
    // Store for early file selection (before main app loads)
    window.__tempFile = null;
    window.__tempFileLoading = false;

    window.__inputFileHandle = null; // FileSystemFileHandle when showOpenFilePicker is used

    function _finishFileSelection(file, handle) {
        window.__inputFileHandle = handle;
        window.__tempFile = file;
        window.__tempFileLoading = true;

        var button = document.getElementById('input-button');
        if (button) {
            button.disabled = true;
            button.classList.add('opacity-60', 'cursor-not-allowed');
            button.innerHTML = '<span class="inline-block animate-spin mr-2">⏳</span> Loading...';
        }

        if (typeof window.processEarlyFile === 'function') {
            window.processEarlyFile(file);
        }
    }

    // Temporary file chooser that works immediately
    window.tempChooseFile = async function() {
        if (window.__tempFileLoading) return;

        // Try showOpenFilePicker first — gives us a FileSystemFileHandle for the resumable pipeline
        if ('showOpenFilePicker' in window) {
            try {
                var picks = await window.showOpenFilePicker({
                    types: [{
                        description: 'Video or image files',
                        accept: {
                            'video/*': ['.mp4', '.webm', '.mov', '.avi', '.mkv', '.ts', '.mts', '.m2ts'],
                            'image/*': ['.jpg', '.jpeg', '.png'],
                        },
                    }],
                    multiple: false,
                });
                var handle = picks[0];
                var file = await handle.getFile();
                _finishFileSelection(file, handle);
                return;
            } catch (e) {
                if (e.name === 'AbortError') return; // user cancelled
                // other error: fall through to input element
            }
        }

        // Fallback: plain file input
        var input = document.createElement('input');
        input.type = 'file';
        input.accept = '.mp4,.mov,.avi,.mkv,.webm,.ts,.mts,.m2ts,image/jpeg,image/png';
        input.style.display = 'none';
        document.body.append(input);
        input.onchange = function(e) {
            var f = e.target.files[0];
            if (f) _finishFileSelection(f, null);
        };
        input.click();
    };
</script>

```python
import bpy
import mathutils

def rig_stickman(mesh_obj):
    # Deselect all
    bpy.ops.object.select_all(action='DESELECT')
    
    # Create armature
    armature = bpy.data.armatures.new("StickmanArmature")
    armature_obj = bpy.data.objects.new("StickmanRig", armature)
    bpy.context.collection.objects.link(armature_obj)
    
    # Set to edit mode
    bpy.context.view_layer.objects.active = armature_obj
    bpy.ops.object.mode_set(mode='EDIT')
    
    bones = {
        'Root': (0, 0, 0, None),
        'Spine': (0, 0, 1, 'Root'),
        'Head': (0, 0, 1.5, 'Spine'),
        'UpperArm.L': (0.5, 0, 1.3, 'Spine'),
        'LowerArm.L': (1, 0, 1.3, 'UpperArm.L'),
        'UpperArm.R': (-0.5, 0, 1.3, 'Spine'),
        'LowerArm.R': (-1, 0, 1.3, 'UpperArm.R'),
        'Thigh.L': (0.2, 0, 0.8, 'Root'),
        'Shin.L': (0.2, 0, 0.4, 'Thigh.L'),
        'Thigh.R': (-0.2, 0, 0.8, 'Root'),
        'Shin.R': (-0.2, 0, 0.4, 'Thigh.R')
    }
    
    created_bones = {}
    
    # Add bones
    for name, (x, y, z, parent_name) in bones.items():
        bone = armature.edit_bones.new(name)
        bone.head = (0, 0, z if name == 'Root' else z - 0.2)
        bone.tail = (x, y, z)
        created_bones[name] = bone
        
    # Parent bones
    for name, bone in created_bones.items():
        parent_name = bones[name][3]
        if parent_name:
            bone.parent = created_bones[parent_name]
            
    bpy.ops.object.mode_set(mode='OBJECT')
    
    # Parent mesh to armature
    mesh_obj.select_set(True)
    armature_obj.select_set(True)
    bpy.context.view_layer.objects.active = armature_obj
    bpy.ops.object.parent_set(type='ARMATURE_AUTO')
    
    return armature_obj
```

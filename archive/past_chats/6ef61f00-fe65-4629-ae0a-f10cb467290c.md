```python
import bpy

def render_video(filepath):
    # Set render engine
    bpy.context.scene.render.engine = 'BLENDER_EEVEE_NEXT' if hasattr(bpy.types, 'BLENDER_EEVEE_NEXT') else 'BLENDER_EEVEE'
    
    # Resolution and FPS
    bpy.context.scene.render.resolution_x = 1080
    bpy.context.scene.render.resolution_y = 1080
    bpy.context.scene.render.fps = 24
    
    # Frame range
    bpy.context.scene.frame_start = 1
    bpy.context.scene.frame_end = 480
    
    # Orthographic camera setup
    cam = bpy.data.cameras.new("OrthoCam")
    cam.type = 'ORTHO'
    cam.ortho_scale = 10.0
    
    cam_obj = bpy.data.objects.new("OrthoCam", cam)
    bpy.context.collection.objects.link(cam_obj)
    bpy.context.scene.camera = cam_obj
    
    # Position camera to look straight at origin
    cam_obj.location = (0, -10, 0)
    cam_obj.rotation_euler = (1.5708, 0, 0) # 90 degrees on X
    
    # Output settings
    bpy.context.scene.render.image_settings.file_format = 'FFMPEG'
    bpy.context.scene.render.ffmpeg.format = 'MPEG4'
    bpy.context.scene.render.ffmpeg.codec = 'H264'
    bpy.context.scene.render.filepath = filepath
    
    # Render
    bpy.ops.render.render(animation=True)
```

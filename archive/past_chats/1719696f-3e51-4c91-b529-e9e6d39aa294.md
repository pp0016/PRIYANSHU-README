```python
import bpy
import math

def animate_stickman(armature):
    """
    Animates a standard stickman armature:
    Frames 1-240: Walk cycle
    Frames 241-480: Laughing fit
    """
    bpy.context.view_layer.objects.active = armature
    bpy.ops.object.mode_set(mode='POSE')
    
    # Assuming standard names, adjust if necessary
    bones = {
        'upper_arm.L': armature.pose.bones.get('upper_arm.L'),
        'upper_arm.R': armature.pose.bones.get('upper_arm.R'),
        'thigh.L': armature.pose.bones.get('thigh.L'),
        'thigh.R': armature.pose.bones.get('thigh.R'),
        'spine': armature.pose.bones.get('spine'),
        'head': armature.pose.bones.get('head'),
        'belly': armature.pose.bones.get('belly')
    }

    # --- Walk Cycle (1-240) ---
    # 24 fps, 24 frame cycle = 10 cycles total
    for frame in range(1, 241):
        phase = (frame % 24) / 24 * 2 * math.pi
        
        if bones['upper_arm.L']:
            bones['upper_arm.L'].rotation_euler[0] = math.sin(phase) * 0.5
            bones['upper_arm.L'].keyframe_insert(data_path="rotation_euler", frame=frame)
        if bones['upper_arm.R']:
            bones['upper_arm.R'].rotation_euler[0] = math.sin(phase + math.pi) * 0.5
            bones['upper_arm.R'].keyframe_insert(data_path="rotation_euler", frame=frame)
        if bones['thigh.L']:
            bones['thigh.L'].rotation_euler[0] = math.sin(phase + math.pi) * 0.8
            bones['thigh.L'].keyframe_insert(data_path="rotation_euler", frame=frame)
        if bones['thigh.R']:
            bones['thigh.R'].rotation_euler[0] = math.sin(phase) * 0.8
            bones['thigh.R'].keyframe_insert(data_path="rotation_euler", frame=frame)

    # --- Laughing Fit (241-480) ---
    for frame in range(241, 481):
        laugh_intensity = math.sin((frame - 241) * 0.2)
        
        if bones['spine']:
            bones['spine'].rotation_euler[0] = laugh_intensity * 0.3
            bones['spine'].keyframe_insert(data_path="rotation_euler", frame=frame)
        if bones['head']:
            bones['head'].rotation_euler[0] = math.sin((frame - 241) * 0.5) * 0.2
            bones['head'].keyframe_insert(data_path="rotation_euler", frame=frame)
        if bones['belly']:
            # Hold belly (simplified by slight translation/scale or rotation)
            bones['belly'].rotation_euler[1] = laugh_intensity * 0.1
            bones['belly'].keyframe_insert(data_path="rotation_euler", frame=frame)

    bpy.ops.object.mode_set(mode='OBJECT')
```
